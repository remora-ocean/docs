13/**
14 * One bilinear value from the four coarse values around it, at weights wx and wy measured from
15 * c00 (outside [0, 1] when extrapolating). With a period, the four are first brought within half
16 * a period of c00, and the result is then taken to the image nearest the coarse value with the
17 * largest weight, so each fine value stays on the branch of its nearest coarse point and a
18 * branch cut (the date line) falls between the same points as on the coarse level.
19 *
20 * @param[in ] c00 coarse value at (ic, jc)
21 * @param[in ] c10 coarse value at (ic+1, jc)
22 * @param[in ] c01 coarse value at (ic, jc+1)
23 * @param[in ] c11 coarse value at (ic+1, jc+1)
24 * @param[in ] wx weight of the ic+1 side
25 * @param[in ] wy weight of the jc+1 side
26 * @param[in ] period period of the field, or 0 for a field that does not wrap
27 */
28AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE
29amrex::Real
30staggered_bilinear_value (amrex::Real c00, amrex::Real c10, amrex::Real c01, amrex::Real c11,
31 amrex::Real wx, amrex::Real wy, amrex::Real period)
32{
33 const amrex::Real half = amrex::Real(0.5);
34 const amrex::Real nearest = (wx < half) ? ((wy < half) ? c00 : c01)
35 : ((wy < half) ? c10 : c11);
36 if (period > amrex::Real(0.0)) {
37 c10 = nearest_periodic_image(c10, c00, period);
38 c01 = nearest_periodic_image(c01, c00, period);
39 c11 = nearest_periodic_image(c11, c00, period);
40 }
41 const amrex::Real v = (amrex::Real(1.0) - wx) * (amrex::Real(1.0) - wy) * c00
42 + wx * (amrex::Real(1.0) - wy) * c10
43 + (amrex::Real(1.0) - wx) * wy * c01
44 + wx * wy * c11;
45 return (period > amrex::Real(0.0)) ? nearest_periodic_image(v, nearest, period) : v;
46}

48/**
49 * \brief Bilinear coarse-to-fine interpolation of a 2D field at any horizontal staggering
50 * (rho, u, v or psi points), for grid coordinates.
51 *
52 * The arithmetic reads the coarse field only where a grid file sets it: the domain, plus the
53 * one-cell ring outside a physical edge along a cell-centred direction. Points beyond
54 * extrapolate linearly from the two nearest values, so a coordinate linear in the index is
55 * reproduced exactly. On a NetCDF level 0 the coordinates' ghost cells beyond that ring are
56 * never set, which is why AMReX's cell and node bilinear interpolaters, which read whatever the
57 * coarse ghost cells hold, don't fit; its face interpolater is also piecewise constant across
58 * the face.
59 *
60 * Used through InterpFromCoarseLevel with a non-periodic coarse Geometry (coordinates do not
61 * wrap; see nonperiodic_geometry) and one BCRec per component saying where the ring is: ext_dir
62 * at a physical edge, int_dir across a periodic one (see staggered_bilinear_bcrec). Across a
63 * periodic edge it reads no coarse ghost cell at all, since those hold the wrapped values from
64 * the other side. InterpFromCoarseLevel copies the parent's ghost cells into the coarse patch,
65 * so the parent needs ceil(fine ghosts / ratio) + 1 of them.
66 *
67 * Constructed with a period (360 for a longitude in degrees, 2 pi for an angle in radians),
68 * it interpolates a field that wraps (a longitude across the date line) across the cut rather
69 * than through the jump, and keeps each fine value on its nearest coarse point's branch (see
70 * staggered_bilinear_value).
71 */
72class StaggeredBilinear
73 : public amrex::Interpolater
74{
75public:
76 /**
77 * @param[in ] period period of the field, or 0 for a field that does not wrap
78 */
79 explicit StaggeredBilinear (amrex::Real period = amrex::Real(0.0)) : m_period(period) {}

108 * Fill fine_region of fine by bilinear interpolation of crse, extrapolating linearly past
109 * the coarse points a grid file sets
110 *
111 * @param[in ] crse coarse patch, as InterpFromCoarseLevel fills it
112 * @param[in ] crse_comp first coarse component
113 * @param[inout] fine fine FAB
114 * @param[in ] fine_comp first fine component
115 * @param[in ] ncomp number of components
116 * @param[in ] fine_region fine points to fill
117 * @param[in ] ratio refinement ratio
118 * @param[in ] crse_geom coarse Geometry; must not be periodic
119 * @param[in ] bcr per component: ext_dir at a physical edge, int_dir across a
120 * periodic one
121 * @param[in ] runon where to run
122 */
123 void interp (const amrex::FArrayBox& crse, int crse_comp, amrex::FArrayBox& fine,
124 int fine_comp, int ncomp, const amrex::Box& fine_region,
125 const amrex::IntVect& ratio, const amrex::Geometry& crse_geom,
126 const amrex::Geometry& /*fine_geom*/, amrex::Vector<amrex::BCRec> const& bcr,
127 int /*actual_comp*/, int /*actual_state*/, amrex::RunOn runon) override
128 {
129 // A periodic copy would give a ghost cell beyond a wall two sources at the seam
130 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(!crse_geom.isAnyPeriodic(),
131 "StaggeredBilinear needs a non-periodic coarse Geometry (nonperiodic_geometry)");
132
133 const amrex::IntVect stag = fine_region.ixType().toIntVect();
134 const amrex::Real sx = amrex::Real(0.5) * amrex::Real(1 - stag[0]);
135 const amrex::Real sy = amrex::Real(0.5) * amrex::Real(1 - stag[1]);
136 const int rx = ratio[0], ry = ratio[1];
137
138 for (int n = 0; n < ncomp; ++n) {
139 amrex::Box filled = amrex::convert(amrex::makeSlab(crse_geom.Domain(), 2, 0), stag);
140 for (int dir = 0; dir < 2; ++dir) {
141 for (const int side : {bcr[n].lo(dir), bcr[n].hi(dir)}) {
142 AMREX_ALWAYS_ASSERT_WITH_MESSAGE(side == amrex::BCType::ext_dir ||
143 side == amrex::BCType::int_dir,
144 "StaggeredBilinear takes ext_dir or int_dir boundary records only");
145 }
146 if (stag[dir] != 0) { continue; }
147 if (bcr[n].lo(dir) == amrex::BCType::ext_dir) { filled.growLo(dir, 1); }
148 if (bcr[n].hi(dir) == amrex::BCType::ext_dir) { filled.growHi(dir, 1); }
149 }
150 filled &= crse.box();
151 AMREX_ALWAYS_ASSERT(filled.length(0) >= 2 && filled.length(1) >= 2);
152 const int ilo = filled.smallEnd(0), ihi = filled.bigEnd(0) - 1;
153 const int jlo = filled.smallEnd(1), jhi = filled.bigEnd(1) - 1;
154
155 amrex::Array4<amrex::Real const> const& c = crse.const_array(crse_comp + n);
156 amrex::Array4<amrex::Real> const& f = fine.array(fine_comp + n);
157 const amrex::Real period = m_period;
158
159 // Written out rather than AMREX_HOST_DEVICE_PARALLEL_FOR_3D_FLAG, whose GPU expansion
160 // names RunOn and Gpu without the amrex:: this header can't assume
161 auto kernel = [=] AMREX_GPU_HOST_DEVICE (int i, int j, int k) noexcept
162 {
163 // Position in the coarse index space, measured from its first point
164 const amrex::Real tx = (amrex::Real(i) + sx) / amrex::Real(rx) - sx;
165 const amrex::Real ty = (amrex::Real(j) + sy) / amrex::Real(ry) - sy;
166 const int ic = amrex::min(amrex::max(static_cast<int>(std::floor(tx)), ilo), ihi);
167 const int jc = amrex::min(amrex::max(static_cast<int>(std::floor(ty)), jlo), jhi);
168 const amrex::Real wx = tx - amrex::Real(ic);
169 const amrex::Real wy = ty - amrex::Real(jc);
170 f(i,j,k) = staggered_bilinear_value(c(ic,jc,k), c(ic+1,jc,k), c(ic,jc+1,k),
171 c(ic+1,jc+1,k), wx, wy, period);
172 };
173 if (runon == amrex::RunOn::Device && amrex::Gpu::inLaunchRegion()) {
174 amrex::ParallelFor(fine_region, kernel);
175 } else {
176 amrex::LoopConcurrentOnCpu(fine_region, kernel);
177 }
178 }
179 }
180
181private:
182 amrex::Real m_period;
183};

185/**
186 * The boundary records StaggeredBilinear expects for a coarse level: ext_dir at a physical
187 * edge, where a grid file supplies a ring, and int_dir across a periodic one. Build them with
188 * this rather than by hand: InterpFromCoarseLevel turns every side of a patch that doesn't reach
189 * the domain edge into int_dir before the interpolater sees it, so the interpolater's own check
190 * catches a wrong record only on patches at the edge.
191 *
192 * @param[in ] crse_geom the coarse level's Geometry, with its real periodicity
193 * @param[in ] ncomp number of components
194 */
195inline amrex::Vector<amrex::BCRec>
196staggered_bilinear_bcrec (const amrex::Geometry& crse_geom, int ncomp = 1)
197{
198 amrex::Vector<amrex::BCRec> bcs(ncomp);
199 for (auto& bc : bcs) {
200 for (int dir = 0; dir < AMREX_SPACEDIM; ++dir) {
201 const int type = crse_geom.isPeriodic(dir) ? amrex::BCType::int_dir
202 : amrex::BCType::ext_dir;
203 bc.setLo(dir, type);
204 bc.setHi(dir, type);
205 }
206 }
207 return bcs;
208}
